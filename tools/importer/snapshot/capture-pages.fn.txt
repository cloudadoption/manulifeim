async (page) => {
  // Captures queued pages one every minGap-maxGap ms; in the gaps it downloads queued images
  // in the page (same origin). Returns {pages, images} for ingest.mjs.
  page.setDefaultTimeout(15000);
  const q = await (await page.request.get('http://localhost:3000/tools/importer/snapshots/_queue.json', { timeout: 5000 })).json();
  // warm session: homepage + attestation first when the browser is not on the site yet
  if (!page.url().startsWith('https://www.manulifeim.com/')) {
    const w = await page.goto('https://www.manulifeim.com/institutional/tw/en', { waitUntil: 'domcontentloaded', timeout: 30000 });
    if (!w || w.status() !== 200) return { pages: [{ url: 'https://www.manulifeim.com/institutional/tw/en', status: w ? w.status() : 0, blocked: true, at: new Date().toISOString() }], images: [] };
    await page.waitForLoadState('load', { timeout: 20000 }).catch(() => {});
    const agree = page.getByRole('button', { name: 'I agree' });
    if (await agree.isVisible().catch(() => false)) await agree.click();
    const reject = page.getByRole('button', { name: 'Reject optional cookies' });
    await reject.waitFor({ state: 'visible', timeout: 8000 }).catch(() => {});
    if (await reject.isVisible().catch(() => false)) await reject.click();
    await page.waitForTimeout(20000);
  }
  const out = [];
  const imgs = [];
  const imgQueue = [...(q.images || [])];
  const t0 = Date.now();
  let errors = 0;
  let imgBlocked = false;
  let img403 = 0;
  for (const url of q.urls) {
    if (Date.now() - t0 > q.budgetMs) break;
    const started = Date.now();
    const rec = { url, at: new Date().toISOString() };
    try {
      const resp = await page.goto(url, { waitUntil: 'domcontentloaded', timeout: 30000 });
      rec.status = resp ? resp.status() : 0;
      rec.finalUrl = page.url();
      const chain = [];
      let r = resp && resp.request().redirectedFrom();
      while (r) { chain.push(r.url()); r = r.redirectedFrom(); }
      rec.redirectedFrom = chain;
      await page.waitForLoadState('load', { timeout: 20000 }).catch(() => {});
      await page.waitForTimeout(2000);
      rec.title = await page.title();
      rec.rendered = await page.evaluate(() => {
        const d = document.documentElement.cloneNode(true);
        d.querySelectorAll('script, noscript, iframe, link[rel=preload], link[rel=prefetch], link[rel=modulepreload]').forEach((e) => e.remove());
        return `<!DOCTYPE html>\n${d.outerHTML}`;
      });
      errors = 0;
    } catch (e) {
      rec.error = String(e.message || e).slice(0, 300);
      errors += 1;
    }
    if ([403, 429, 503].includes(rec.status) || /access denied/i.test(rec.title || '') || errors >= 2) {
      rec.blocked = true;
      delete rec.rendered;
      out.push(rec);
      break;
    }
    out.push(rec);
    const gap = (q.minGapMs || 5000) + Math.random() * ((q.maxGapMs || 10000) - (q.minGapMs || 5000));
    // images in the gap, 1.5 s apart, until the gap is used up
    while (!imgBlocked && rec.status === 200 && imgQueue.length && Date.now() - started < gap - 3000) {
      const iu = imgQueue.shift();
      const got = await page.evaluate(async (u) => {
        try {
          const ctrl = new AbortController();
          const timer = setTimeout(() => ctrl.abort(), 15000);
          const r = await fetch(u, { credentials: 'include', signal: ctrl.signal });
          clearTimeout(timer);
          const blob = await r.blob();
          const b64 = r.ok ? await new Promise((res) => {
            const fr = new FileReader();
            fr.onload = () => res(String(fr.result).split(',')[1]);
            fr.readAsDataURL(blob);
          }) : null;
          return { url: u, status: r.status, type: blob.type, b64 };
        } catch (e) { return { url: u, status: 0, error: String(e.message || e) }; }
      }, iu).catch((e) => ({ url: iu, status: 0, error: String(e.message || e) }));
      imgs.push(got);
      // single assets can be 403 on live itself; three in a row means we are blocked
      img403 = [403, 429].includes(got.status) ? img403 + 1 : 0;
      if (img403 >= 3) imgBlocked = true;
      await page.waitForTimeout(1500);
    }
    const wait = gap - (Date.now() - started);
    if (wait > 0) await page.waitForTimeout(wait);
  }
  return { pages: out, images: imgs };
}
