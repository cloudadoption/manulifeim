async (page) => {
  page.setDefaultTimeout(15000);
  const q = await (await page.request.get('http://localhost:3000/tools/importer/snapshots/_img-queue.json', { timeout: 5000 })).json();
  return page.evaluate(async (urls) => {
    const out = [];
    for (const url of urls) {
      try {
        const ctrl = new AbortController();
        const timer = setTimeout(() => ctrl.abort(), 20000);
        const r = await fetch(url, { credentials: 'include', signal: ctrl.signal });
        clearTimeout(timer);
        const blob = await r.blob();
        const b64 = r.ok ? await new Promise((res) => {
          const fr = new FileReader();
          fr.onload = () => res(String(fr.result).split(',')[1]);
          fr.readAsDataURL(blob);
        }) : null;
        out.push({ url, status: r.status, type: blob.type, b64 });
        if ([403, 429].includes(r.status)) break;
      } catch (e) {
        out.push({ url, status: 0, error: String(e.message || e) });
      }
      await new Promise((res) => { setTimeout(res, 1500); });
    }
    return out;
  }, q.urls);
}
