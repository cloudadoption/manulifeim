async (page) => {
  // Published-look regression check for code changes (Playwright MCP browser_run_code).
  // Job: tools/visual-diff/_job.json on the dev server:
  //   { "a": "https://main--manulifeim--cloudadoption.aem.page", "b": "https://<branch>--...",
  //     "paths": ["/institutional/tw/en", ...], "widths": [1440, 768, 390], "tolerance": 2 }
  // For each path and width it loads a, then b (one page at a time), records every visible
  // h1-h6 / p / li / a / img / button in <main> and <header>/<footer> (box, font size, colour,
  // text start) and reports elements that moved, resized, restyled, appeared or disappeared.
  page.setDefaultTimeout(15000);
  const job = await (await page.request.get('http://localhost:3000/tools/visual-diff/_job.json', { timeout: 5000 })).json();
  const tol = job.tolerance ?? 2;
  const collect = () => {
    const out = [];
    document.querySelectorAll('main h1, main h2, main h3, main h4, main h5, main h6, main p, main li, main a, main img, main button, footer a, footer p')
      .forEach((el) => {
        const r = el.getBoundingClientRect();
        if (!r.width || !r.height) return;
        const cs = getComputedStyle(el);
        if (cs.visibility === 'hidden') return;
        const text = (el.tagName === 'IMG' ? (el.getAttribute('alt') || el.currentSrc.split('/').pop().split('?')[0]) : el.textContent).replace(/\s+/g, ' ').trim().slice(0, 40);
        out.push({
          k: `${el.tagName}|${text}`,
          x: Math.round(r.x), y: Math.round(r.y + window.scrollY), w: Math.round(r.width), h: Math.round(r.height),
          f: `${cs.fontSize}/${cs.fontWeight}/${cs.color}`,
        });
      });
    return { items: out, height: document.documentElement.scrollHeight };
  };
  const load = async (base, path, width) => {
    await page.setViewportSize({ width, height: 900 });
    await page.goto(`${base}${path}`, { waitUntil: 'load', timeout: 30000 });
    await page.evaluate(() => localStorage.setItem('mim-inst-attestation-accepted-TW', 'true'));
    await page.reload({ waitUntil: 'load', timeout: 30000 });
    // scroll through so lazy blocks/images load, then back to the top
    await page.evaluate(async () => {
      for (let y = 0; y < document.documentElement.scrollHeight; y += 700) { window.scrollTo(0, y); await new Promise((r) => setTimeout(r, 120)); }
      window.scrollTo(0, 0);
    });
    await page.waitForTimeout(2500);
    return page.evaluate(collect);
  };
  const report = [];
  for (const path of job.paths) {
    for (const width of job.widths) {
      const A = await load(job.a, path, width);
      const B = await load(job.b, path, width);
      const byKey = (list) => { const m = new Map(); list.forEach((i) => { if (!m.has(i.k)) m.set(i.k, []); m.get(i.k).push(i); }); return m; };
      const ma = byKey(A.items); const mb = byKey(B.items);
      const diffs = [];
      ma.forEach((as, k) => {
        const bs = mb.get(k) || [];
        as.forEach((a, n) => {
          const b = bs[n];
          if (!b) { diffs.push(`- gone ${k} @${a.x},${a.y} ${a.w}x${a.h}`); return; }
          const moved = Math.abs(a.x - b.x) > tol || Math.abs(a.y - b.y) > tol;
          const sized = Math.abs(a.w - b.w) > tol || Math.abs(a.h - b.h) > tol;
          if (moved || sized || a.f !== b.f) {
            diffs.push(`~ ${k} ${a.x},${a.y} ${a.w}x${a.h} -> ${b.x},${b.y} ${b.w}x${b.h}${a.f !== b.f ? ` style ${a.f} -> ${b.f}` : ''}`);
          }
        });
      });
      mb.forEach((bs, k) => { const as = ma.get(k) || []; bs.slice(as.length).forEach((b) => diffs.push(`+ new ${k} @${b.x},${b.y} ${b.w}x${b.h}`)); });
      report.push({ path, width, heightA: A.height, heightB: B.height, items: A.items.length, diffs: diffs.length, first: diffs.slice(0, job.max ?? 8) });
    }
  }
  return report;
}
